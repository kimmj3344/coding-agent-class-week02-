/* ---------------- grading (A안) ---------------- */
function normMean(s){
  return String(s||'').toLowerCase().trim()
    .replace(/[.,!?;:'"()\[\]{}·…~\-_/\\＋+]/g,'')
    .replace(/\s+/g,'');
}
function normSpell(s){ return String(s||'').toLowerCase().trim().replace(/\s+/g,' '); }

function lev(a,b){
  if(a===b) return 0;
  if(!a.length) return b.length;
  if(!b.length) return a.length;
  var prev=new Array(b.length+1), cur=new Array(b.length+1), i,j;
  for(j=0;j<=b.length;j++) prev[j]=j;
  for(i=1;i<=a.length;i++){
    cur[0]=i;
    for(j=1;j<=b.length;j++){
      cur[j]=Math.min(prev[j]+1, cur[j-1]+1, prev[j-1]+(a[i-1]===b[j-1]?0:1));
    }
    var tmp=prev; prev=cur; cur=tmp;
  }
  return prev[b.length];
}
function sim(a,b){ var m=Math.max(a.length,b.length); return m?1-lev(a,b)/m:0; }

// returns {level:'exact'|'close'|'wrong', match:<가장 가까운 등록된 뜻>}
function gradeMeaning(input, meanings){
  var u=normMean(input);
  if(!u) return {level:'wrong', match:meanings[0]};
  var best={level:'wrong', score:0, match:meanings[0]};
  for(var i=0;i<meanings.length;i++){
    var t=normMean(meanings[i]);
    if(!t) continue;
    if(u===t) return {level:'exact', match:meanings[i]};
    // 한쪽이 다른 쪽을 품으면("사고" ↔ "사고하다") 어미·수식어 차이로 보고 넉넉히 인정
    var contained = (t.length>=2 && u.length>=2 && (t.indexOf(u)>=0 || u.indexOf(t)>=0));
    var ratio = contained ? Math.min(u.length,t.length)/Math.max(u.length,t.length) : 0;
    var s = sim(u,t);
    if(contained && ratio>=0.4){ var cs=0.65+ratio*0.3; if(cs>s) s=cs; }
    // 어간이 같고 어미만 다르면("유창한" ↔ "유창하다") 인정
    var p=0, lim=Math.min(u.length,t.length);
    while(p<lim && u[p]===t[p]) p++;
    if(p>=2 && p/lim>=0.6 && s<0.72) s=0.72;
    if(s>=0.6 && s>best.score) best={level:'close', score:s, match:meanings[i]};
  }
  return best;
}
function gradeSpelling(input, word){
  var u=normSpell(input), t=normSpell(word);
  if(u===t) return {level:'exact', match:word};
  if(t.length>=4 && lev(u,t)===1) return {level:'wrong', near:true, match:word};
  return {level:'wrong', match:word};
}

function mastery(w){

}

// words[]는 스냅샷마다 새 객체로 다시 만들어진다. 비동기 작업을 가로질러
// 단어를 들고 다닐 때는 반드시 id로 다시 집을 것 — 객체 비교는 어긋난다.
function wordById(id){
  for(var i=0;i<words.length;i++) if(words[i].id===id) return words[i];
  return null;
}
function qWord(){ return wordById(quiz.queue[quiz.i]); }

function buildChoices(target){
  var correct=target.meanings[0]||'';
  var seen={}; seen[normMean(correct)]=1;
  var out=[];
  // 같은 단어장 안에서 먼저 보기를 고르고, 모자라면 다른 단어장에서 채운다
  var same=shuffle(words.filter(function(w){return w.id!==target.id && w.deck===target.deck;}));
  var rest=shuffle(words.filter(function(w){return w.id!==target.id && w.deck!==target.deck;}));
  var others=same.concat(rest);
  for(var i=0;i<others.length && out.length<3;i++){
    var m=others[i].meanings[0]; if(!m) continue;
    var k=normMean(m); if(seen[k]) continue;
    seen[k]=1; out.push(m);
  }
  if(out.length<3) return null;
  return shuffle([correct].concat(out));
}

  typef.hidden=false; input.className=''; input.value='';
  input.placeholder='뜻'; input.focus();

function answerChoice(picked, w, all){
  if(quiz.answered) return;
  var ok = normMean(picked)===normMean(w.meanings[0]);
  var btns=$('#q-choices').children;
  for(var i=0;i<btns.length;i++){
    var b=btns[i];
    b.disabled=true;
    var txt=b.lastChild.textContent;
    if(normMean(txt)===normMean(w.meanings[0])) b.className='ch right';
    else if(txt===picked) b.className='ch bad';
  }
  resolve(ok?'exact':'wrong', w, picked, w.meanings[0], false);
}

function submitTyped(e){
  e.preventDefault();
  if(!quiz || quiz.answered || quiz.checking) return;
  var w=qWord(); if(!w) return;
  var v=$('#q-input').value;
  if(!v.trim()) return;

  if(quiz.mode==='spelling'){
    var gs=gradeSpelling(v,w.word);
    resolve(gs.level, w, v, gs.match, !!gs.near);
    return;
  }
  // A안: 등록된 뜻으로 먼저 채점 — 통과하면 네트워크 없이 즉시 판정
  var g=gradeMeaning(v,w.meanings);
  if(g.level!=='wrong'){ resolve(g.level, w, v, g.match, false); return; }
  // 여기서만 AI에게 물어본다
  if(aiOn()){ aiCheck(w, v); return; }
  resolve('wrong', w, v, g.match, false, aiOffNote());
}

  quiz.checking=true;

  var fb=$('#q-fb'); fb.hidden=false; fb.className='fb close'; fb.innerHTML='';

  fb.appendChild(el('span','fm','등록된 뜻과 글자가 달라서 AI에게 물어보고 있어요. 보통 1~2초 걸립니다.'));

  var targetId=w.id, answer=given;
  // 응답이 돌아왔을 때도 같은 문제에 머물러 있는지 id로 확인한다.
  function stillHere(){
    return quiz && !quiz.answered && quiz.phase==='play' && quiz.queue[quiz.i]===targetId && wordById(targetId);
  }
  // 기본 호출만 쓴다. sample.json은 일부 뷰어에 아예 없어서, 없으면
  // 호출 순간 동기적으로 터지고 .catch에도 걸리지 않는다.
  var p;
  try{
    p = (typeof aiCap==='function') ? aiCap(aiPrompt(w, given), {modelTier:'quick'}) : null;
  }catch(err){
    p = Promise.reject((err && err.code) ? err : {code:'upstream_error', message:String(err)});
  }
  if(!p || typeof p.then!=='function'){
    p = Promise.reject({code:'capability_removed', message:'sample() unavailable'});
  }

  p.then(function(r){
    endCheck();
    var target=stillHere(); if(!target) return;
    var txt = String((r && r.text) || '').trim();
    var head = (txt.split(/[\s\n]+/)[0] || '').toUpperCase().replace(/[^A-Z]/g,'');
    var same = head.indexOf('SAME')===0;
    var why = txt.split('\n').slice(1).join(' ').trim().slice(0,120);
    if(same){
      // 단어장에는 넣지 않는다 — 등록된 뜻은 사용자가 직접 적은 것만 남긴다
      creditClose(target, answer,
        '같은 뜻이네요 — 정답',
        (why?esc(why)+'<br>':'') + '등록된 뜻 → <b>'+esc(target.meanings.join(', '))+'</b>',
        false);
    } else {
      resolve('wrong', target, answer, target.meanings[0], false, why?('AI 판정: '+why):'');
    }
  }).catch(function(e){
    endCheck();
    var target=stillHere(); if(!target) return;
    var c=(e&&e.code)||'upstream_error';
    var note;
    if(c==='not_granted'){
      aiResolved=true; aiAvail=false; syncAiRow();
      note='이 페이지가 Claude를 쓰는 걸 허용하지 않아서 등록된 뜻으로만 채점했어요. ('+c+')';
    } else if(c==='sampling_disabled'||c==='not_declared'||c==='capability_disabled'||c==='capability_removed'){
      aiResolved=true; aiAvail=false; syncAiRow();
      note='이 계정/화면에서는 AI 판정을 쓸 수 없어서 등록된 뜻으로만 채점했어요. ('+c+')';
    } else if(c==='rate_limited'){
      note='AI 호출이 잠시 막혀서 등록된 뜻으로만 채점했어요. 잠시 뒤에 다시 됩니다. ('+c+')';
    } else if(c==='cancelled'){
      note='';
    } else {
      note='AI 판정에 실패해서 등록된 뜻으로만 채점했어요. ('+c+')';
    }
    resolve('wrong', target, answer, target.meanings[0], false, note);
  });

  function endCheck(){
    if(quiz) quiz.checking=false;
    var inp=$('#q-input'); if(inp) inp.disabled=false;
    var b=$('#q-typef').querySelector('button'); if(b) b.disabled=false;
  }
}

function resolve(level, w, given, match, near, aiNote){
  var base = level==='exact'?10 : level==='close'?7 : 0;
  var gained=0;
  var fb=$('#q-fb'); fb.innerHTML=''; fb.hidden=false;

  if(level==='wrong'){
    quiz.combo=0; quiz.wrongN++;
    quiz.misses.push({word:w.word, answer:w.meanings.join(', '), given:given});
    fb.className='fb bad';
    var t=el('span','ft','오답');
    var m=el('span','fm');
    if(quiz.mode==='spelling'){
      m.innerHTML = (near?'한 글자 차이였어요. ':'') + '정답은 <b>'+esc(w.word)+'</b>';
    } else {
      m.innerHTML = '정답은 <b>'+esc(w.meanings.join(', '))+'</b>';
    }
    if(aiNote) m.innerHTML += '<br><span style="color:var(--muted)">'+esc(aiNote)+'</span>';
    fb.appendChild(t); fb.appendChild(m);
    quiz.answered={level:level, w:w, given:given, delta:{wrong:1}};
  } else {
    quiz.combo++; quiz.right++;
    if(quiz.combo>quiz.maxCombo) quiz.maxCombo=quiz.combo;
    gained=Math.round(base*multiplier(quiz.combo-1));
    quiz.score+=gained;
    fb.className='fb '+(level==='exact'?'good':'close');
    fb.appendChild(el('span','ft', (level==='exact'?'정답':'비슷해요 — 정답으로 칠게요')+' +'+gained+multLabel(quiz.combo-1)));
    var m2=el('span','fm');
    if(level==='close'){
      m2.innerHTML='이게 더 정확해요 → <b>'+esc(w.meanings.join(', '))+'</b>';
    } else if(quiz.mode==='spelling'){
      m2.innerHTML='<b>'+esc(w.word)+'</b> · '+esc(w.meanings.join(', '));
    } else {
      m2.textContent=w.meanings.join(', ');
    }
    fb.appendChild(m2);
    quiz.answered={level:level, w:w, given:given, delta:{right:1}};
  }

  updateHud();

  var acts=el('div','fb-acts');
  if(level==='wrong' && quiz.mode!=='spelling' && given && given.trim()){
    var rescue=el('button','btn','“'+(given.trim().length>14?given.trim().slice(0,14)+'…':given.trim())+'” 도 맞는 뜻이에요');
    rescue.type='button';
    rescue.addEventListener('click', function(){ rescueAnswer(w, given.trim()); });
    acts.appendChild(rescue);
  }
  var nxt=el('button','btn primary', quiz.i+1>=quiz.queue.length?'결과 보기':'다음 문제 →');
  nxt.type='button';
  nxt.addEventListener('click', advance);
  acts.appendChild(nxt);
  fb.appendChild(acts);
  nxt.focus();
}

function updateHud(){
  if(!quiz) return;
  $('#q-score').textContent=quiz.score;
  $('#q-combo').textContent=quiz.combo;
  $('#q-combo').dataset.hot = quiz.combo>=5?'1':'0';
  $('#q-combofill').style.width=Math.min(100,(quiz.combo/10)*100)+'%';
}

function saveMeaningTo(w, given){
  var cur=wordById(w.id)||w;
  var g=String(given||'').trim(); if(!g) return;
  var mset=cur.meanings.slice(), k=normMean(g);
  for(var i=0;i<mset.length;i++) if(normMean(mset[i])===k) return;
  mset.push(g);
  patch(cur.id,{meanings:mset});
}

// '비슷함'과 같은 대우로 정답 처리 (7점). AI 인정과 수동 구제가 함께 쓴다.
function creditClose(w, given, headline, noteHtml, save){
  if(save) saveMeaningTo(w, given);
  quiz.combo++; quiz.right++;
  if(quiz.combo>quiz.maxCombo) quiz.maxCombo=quiz.combo;
  var gained=Math.round(7*multiplier(quiz.combo-1));
  quiz.score+=gained;
  quiz.answered={level:'close', w:w, given:given, delta:{right:1}};

  var fb=$('#q-fb'); fb.hidden=false; fb.className='fb close'; fb.innerHTML='';
  fb.appendChild(el('span','ft', headline+' +'+gained+multLabel(quiz.combo-1)));
  var m=el('span','fm'); m.innerHTML=noteHtml; fb.appendChild(m);

  var acts=el('div','fb-acts');
  var nxt=el('button','btn primary', quiz.i+1>=quiz.queue.length?'결과 보기':'다음 문제 →');
  nxt.type='button'; nxt.addEventListener('click', advance);
  acts.appendChild(nxt); fb.appendChild(acts);

  updateHud();
  nxt.focus();
}

function rescueAnswer(w, given){
  quiz.misses.pop();
  quiz.wrongN--;
  creditClose(w, given, '정답으로 바꿨어요',
    '<b>'+esc(given)+'</b> 을(를) '+esc(w.word)+'의 뜻으로 저장했어요. 다음부터는 바로 정답 처리됩니다.',
    true);
}

function advance(){
  var a=quiz.answered;
  if(a){
    var w=wordById(a.w.id)||a.w;   // 최신 집계를 보고 더한다
    if(a.delta.wrong) patch(w.id,{wrong:(w.wrong||0)+1});
    else patch(w.id,{right:(w.right||0)+1});
  }
  quiz.i++;
  if(quiz.i>=quiz.queue.length) finishQuiz(); else nextQuestion();
}

function finishQuiz(){
  quiz.phase='done';
  var box=$('#q-done'); box.innerHTML='';
  var total=quiz.right+quiz.wrongN;
  box.appendChild(el('div','big-score', String(quiz.score)));
  box.appendChild(el('p','sub', total? (total+'문제 중 '+quiz.right+'개 정답') : '문제를 풀지 않았어요'));

  var g=el('div','rgrid');
  [['정답',quiz.right],['오답',quiz.wrongN],['최고 콤보',quiz.maxCombo]].forEach(function(p){
    var d=el('div');
    d.appendChild(el('span','hl',p[0]));
    d.appendChild(el('span','hv',String(p[1])));
    g.appendChild(d);
  });
  box.appendChild(g);

  if(quiz.misses.length){
    var r=el('div','rmiss');
    r.appendChild(el('div','sec-title','오답노트에 담긴 단어'));
    var ul=el('ul');
    quiz.misses.forEach(function(m){
      var li=el('li');
      li.appendChild(el('span',null,m.word));
      li.appendChild(el('span',null,m.answer));
      ul.appendChild(li);
    });
    r.appendChild(ul); box.appendChild(r);
  }

  var acts=el('div'); acts.style.cssText='display:flex;gap:8px;flex-wrap:wrap';
  var again=el('button','btn primary','다시 풀기'); again.style.flex='1';
  again.addEventListener('click', function(){ quiz=null; renderQuiz(); });
  acts.appendChild(again);
  if(quiz.misses.length){
    var mq=el('button','btn','오답만 다시'); mq.style.flex='1';
    mq.addEventListener('click', function(){ quiz=null; qScope='wrong'; syncScopeChips(); syncQDeck(); renderQuiz(); startQuiz(); });
    acts.appendChild(mq);
  }
  box.appendChild(acts);
  renderQuiz();
  render();
}
